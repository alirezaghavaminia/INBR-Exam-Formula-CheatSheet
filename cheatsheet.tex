\begin{secbox}{Humidity Ratio}{نسبت رطوبت}
\ptext{جرم بخار آب به‌ازای واحد جرم هوای خشک؛ ضریب $0.622$ نسبت جرم مولکولی آب به هواست \pa{$18.02/28.97$}. در IP معمولاً با واحد grain/lb کار می‌شود \pa{$1\ \mathrm{lb}=7000\ \mathrm{gr}$}.}
\ver{SI}{\omega=\frac{m_v}{m_a}=0.622\,\frac{P_v}{P-P_v}\ \ [\mathrm{kg/kg}], \omega\,[\mathrm{g/kg}]=1000\,\omega\,[\mathrm{kg/kg}]}
\ver{IP}{\omega=\frac{m_v}{m_a}=0.622\,\frac{P_v}{P-P_v}\ \ [\mathrm{lb/lb}], W\,[\mathrm{gr/lb}]=7000\,\omega\,[\mathrm{lb/lb}]}
\ver{SI, IP}{\gamma=\frac{m_v}{m_v+m_a}=\frac{\omega}{1+\omega}\ \ [\mathrm{kg/kg},\ \mathrm{lb/lb}]}

\begin{secbox}{Relative Humidity}{رطوبت نسبی}
\ptext{نسبت فشار بخار موجود به فشار بخار اشباع در همان دمای خشک؛ برای تبدیل به $\omega$ از رابطهٔ زیر استفاده کنید. $P_g$ از جدول فشار اشباع می‌آید و $P_v$ و $P$ باید هم‌واحد باشند.}
\ver{SI/IP}{\varphi=\frac{P_v}{P_g(T_{db})}\,,\qquad \omega=0.622\,\frac{\varphi P_g}{P-\varphi P_g}}
\ver{SI/IP}{\mu=\frac{\omega}{\omega_s}\quad(\text{degree of saturation}),\qquad \omega_s=0.622\,\frac{P_g}{P-P_g}}
\par\vspace{1pt}{\small\noindent
\begin{tabular*}{\linewidth}{@{}l@{\hspace{1.2em}}l@{\hspace{0.4em}}l@{\hspace{1.2em}}l@{\hspace{0.4em}}l@{\extracolsep{\fill}}r@{}}
$\varphi$         & \multicolumn{4}{l}{{\color{redn}$\text{--}$}}                                                   & \fa{رطوبت نسبی \pa{کسر یا درصد}}\\
$\mu$             & \multicolumn{4}{l}{{\color{redn}$\text{--}$}}                                                   & \fa{درجهٔ اشباع}\\
$P_v,P_g,P$       & {\color{redn}\textsf{\bfseries SI}} & {\color{redn}$\mathrm{kPa}$} & {\color{redn}\textsf{\bfseries IP}} & {\color{redn}$\mathrm{psia}$} & \fa{فشار بخار، فشار اشباع در دمای خشک، فشار کل}\\
$\omega,\omega_s$ & \multicolumn{4}{l}{{\color{redn}$\mathrm{kg/kg}$ \ \ ($\mathrm{lb/lb}$)}}                        & \fa{نسبت رطوبت واقعی و اشباع}\\

\end{secbox}
